import { connect } from "node:net";
import { Worker, isMainThread, workerData, parentPort } from "node:worker_threads";
import { Session } from "node:inspector";

const disabled = /--(?:disable-sigusr1|experimental-permission|permission)\b/.test(
  [...process.execArgv, process.env.NODE_OPTIONS ?? ""].join(" ")
);
const supported = !disabled && !process.versions.bun && process.platform !== "win32" && process.getBuiltinModule;
const address = process.env.POLARIS_PROFILE_SOCKET;
const token = process.env.POLARIS_PROFILE_TOKEN;

function channel(onMessage, onClose) {
  const socket = connect(address);
  let buffer = "";
  socket.on("error", () => socket.destroy());
  socket.on("close", onClose);
  socket.on("data", (chunk) => {
    buffer += chunk;
    if (buffer.length > 16384) return socket.destroy();
    let end;
    while ((end = buffer.indexOf("\n")) >= 0) {
      const line = buffer.slice(0, end);
      buffer = buffer.slice(end + 1);
      try { onMessage(JSON.parse(line)); } catch { socket.destroy(); }
    }
  });
  socket.unref();
  return socket;
}

function register() {
  let worker;
  const socket = channel((message) => {
    if (message.kind !== "capture" || worker || typeof message.nonce !== "string") return;
    worker = new Worker(new URL(import.meta.url), {
      workerData: { [token]: true, nonce: message.nonce, seconds: message.seconds }, execArgv: [],
    });
    worker.on("error", () => {});
    worker.on("exit", () => { worker = undefined; });
    worker.unref();
  }, () => {});
  socket.on("connect", () => socket.write(JSON.stringify({
    kind: "register", token, pid: process.pid, runtime: "node",
  }) + "\n"));
}

function watchdog() {
  const session = new Session();
  let opened = false;
  let closing = false;
  let timeout;
  let lease;
  const send = (message) => socket.write(JSON.stringify({ ...message, pid: process.pid }) + "\n");
  const post = (method, params = {}) => new Promise((resolve, reject) => {
    session.post(method, params, (error, result) => error ? reject(error) : resolve(result));
  });
  const close = () => {
    if (closing) return;
    closing = true;
    clearTimeout(timeout);
    clearTimeout(lease);
    session.disconnect();
    socket.destroy();
    parentPort.close();
  };
  const refresh = () => {
    clearTimeout(lease);
    lease = setTimeout(close, 1500);
  };
  const capture = async () => {
    const check = await post("Runtime.evaluate", {
      expression: "process.getBuiltinModule('node:inspector').url() ?? null", returnByValue: true,
    });
    if (check.exceptionDetails || check.result.value !== null) throw new Error("Existing inspector");
    await post("Profiler.enable");
    await post("Profiler.start");
    if (closing) return;
    send({ kind: "opened" });
    await new Promise(resolve => { timeout = setTimeout(resolve, workerData.seconds * 1000); });
    if (closing) return;
    const { profile } = await post("Profiler.stop");
    if (closing) return;
    const text = JSON.stringify(profile);
    if (Buffer.byteLength(text) > 64 * 1024 * 1024) throw new Error("Profile too large");
    for (let offset = 0; offset < text.length; offset += 8192) {
      if (!send({ kind: "chunk", data: text.slice(offset, offset + 8192) })) {
        await new Promise(resolve => {
          const settled = () => { socket.off("drain", settled); socket.off("close", settled); resolve(); };
          socket.once("drain", settled); socket.once("close", settled);
        });
      }
      if (closing) return;
    }
    send({ kind: "completed" });
  };
  const socket = channel((message) => {
    if (message.kind === "ping") refresh();
    if (message.kind === "close") close();
    if (message.kind !== "open" || opened || closing) return;
    opened = true;
    void capture().catch(() => { if (!closing) send({ kind: "refused" }); close(); });
  }, close);
  session.connectToMainThread();
  socket.ref();
  socket.on("connect", () => {
    send({kind: "watchdog", token, nonce: workerData.nonce});
    refresh();
  });
}

if (supported && address && token) {
  try {
    if (isMainThread) register();
    else if (workerData?.[token] === true) watchdog();
  } catch {}
}
