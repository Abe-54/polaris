import { decode, port, summary } from "./codec.ts";

const capture = process.env.SPIKE_CAPTURE === "1";
const requests: Array<object> = [];
let spans = 0;
let batches = 0;
let server: ReturnType<typeof Bun.serve> | undefined;
if (process.argv[2] !== "bare") {
  server = Bun.serve({
    hostname: "127.0.0.1",
    port,
    reusePort: false,
    maxRequestBodySize: 4 * 1024 * 1024,
    async fetch(req) {
      const path = new URL(req.url).pathname;
      if (path === "/stats" && req.method === "GET") {
        return Response.json({ spans, batches, requests });
      }
      // Drain rejected requests too: Bun 1.3 can otherwise misread the next keep-alive request.
      const bytes = new Uint8Array(await req.arrayBuffer());
      if (!["/v1/traces", "/vendor/v1/traces", "/otlp/spike/v1/traces"].includes(path)) {
        return new Response(null, { status: 404 });
      }
      if (req.method !== "POST") return new Response(null, { status: 405 });
      const type = req.headers.get("content-type")?.split(";")[0] ?? "";
      if (!["application/x-protobuf", "application/json"].includes(type)) {
        return new Response(null, { status: 415 });
      }
      try {
        const result = summary(decode(bytes, type));
        spans += result.spans;
        batches++;
        if (capture) {
          requests.push({ path, type, bytes: bytes.length, sender: req.headers.get("x-spike-sender"), ...result });
          if (req.headers.get("x-spike-sender") === "manual") {
            await Bun.write("fixture.pb", bytes);
          }
        }
        return new Response(type === "application/json" ? "{}" : new Uint8Array(), {
          headers: { "content-type": type },
        });
      } catch {
        return new Response(null, { status: 400 });
      }
    },
  });
}
console.log(JSON.stringify({ ready: true, pid: process.pid, port: server?.port ?? null }));
// stdin holds the bare process open without a timer. The parent closes it on every exit path.
for await (const _ of Bun.stdin.stream()) { /* no commands */ }
await server?.stop(true);
