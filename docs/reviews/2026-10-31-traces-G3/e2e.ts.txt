/** G3 end-to-end: real Node OTel app -> source Daemon receiver -> traces RPCs and MCP HTTP tools. */
import assert from "node:assert/strict";
import { cp, mkdtemp, mkdir, realpath, rm, writeFile } from "node:fs/promises";
import { join } from "node:path";
import { Effect, Layer, Stream } from "effect";

const REPO = "/Users/lucasdoell/code/polaris.worktrees/polaris/grill-with-docs-i-want-to-do-7ce3";
const { ConstellationId } = await import(`${REPO}/packages/protocol/src/index.ts`);
const { EventStore } = await import(`${REPO}/apps/daemon/src/store/EventStore.ts`);
const { seed } = await import(`${REPO}/apps/daemon/src/traces/authority.testing.ts`);
const { TEST_SESSION, TEST_WORKSPACE } = await import(`${REPO}/apps/daemon/src/traces/testing.ts`);
const { McpTokens } = await import(`${REPO}/apps/daemon/src/mcp/tokens.ts`);
const { McpBinding } = await import(`${REPO}/apps/daemon/src/mcp/binding.ts`);
const { launchDaemon, awaitReady, connect } = await import(`${REPO}/packages/bench/src/daemon.ts`);

const results: Array<{ name: string; pass: boolean; detail?: unknown }> = [];
const check = (name: string, pass: boolean, detail?: unknown) => {
  results.push({ name, pass, detail });
  console.log(`${pass ? "PASS" : "FAIL"} ${name}${detail === undefined ? "" : " " + JSON.stringify(detail)}`);
};
const keep = process.env.G3_KEEP === "1";

const home = await mkdtemp("/tmp/g3h-");
const workspace = await realpath(await mkdtemp("/tmp/g3e2e/ws-"));
const appDir = join(workspace, "package");
console.log(JSON.stringify({ home, workspace }));
let daemon: Awaited<ReturnType<typeof launchDaemon>> | undefined;
let app: ReturnType<typeof Bun.spawn> | undefined;

try {
  await mkdir(appDir);
  await cp("/tmp/rpu-p7/node/src", join(appDir, "src"), { recursive: true });
  await cp("/tmp/rpu-p7/node/instrumentation.ts", join(appDir, "instrumentation.ts"));
  await cp("/tmp/rpu-p7/node/package.json", join(appDir, "package.json"));
  await cp("/tmp/rpu-p7/node/node_modules", join(appDir, "node_modules"), { recursive: true, verbatimSymlinks: true });

  await Effect.runPromise(Effect.scoped(seed(workspace).pipe(Effect.provide(EventStore.layerSqlite(join(home, "state.sqlite"))))));
  const binding = McpBinding.cases.Plain.make({ sessionId: TEST_SESSION, constellationId: ConstellationId.make("g3-e2e"), workspaceId: TEST_WORKSPACE });
  const mcpToken: string = await Effect.runPromise(
    Effect.scoped(Effect.flatMap(McpTokens, (t: any) => t.issue(binding)).pipe(Effect.provide(McpTokens.layer(join(home, "mcp.sqlite")))))
  );
  console.log("seeded workspace+session and issued a Plain MCP binding for", TEST_SESSION);

  daemon = await launchDaemon({ binary: null, home });
  console.log("daemon argv:", daemon.argv.join(" "), "pid", daemon.pid);

  await Effect.runPromise(Effect.scoped(Effect.gen(function* () {
    const readyMs = yield* awaitReady(daemon!);
    console.log("daemon ready in ms", Math.round(readyMs));
    const client = yield* connect(daemon!, "socket", "g3-e2e");
    const rpc = client.connection.client as any;
    check("hello advertises traces capability", client.capabilities.includes("traces"), client.capabilities.filter((c: string) => c.startsWith("traces")));

    const firstPage = (filter?: unknown) =>
      Stream.runHead(rpc["traces.watch"](filter === undefined ? { sessionId: TEST_SESSION } : { sessionId: TEST_SESSION, filter })).pipe(
        Effect.map((o: any) => { assert.equal(o._tag, "Some"); return o.value; })
      );

    const initial: any = yield* firstPage();
    console.log("initial watch page:", JSON.stringify({ ...initial, endpoint: initial.endpoint?.replace(/otlp\/[^/]+/, "otlp/<token>") }));
    check("receiver Listening", initial.availability._tag === "Listening", initial.availability);
    check("first watch page carries endpoint", typeof initial.endpoint === "string" && initial.endpoint.startsWith("http://127.0.0.1:4319/otlp/"));
    const endpoint: string = initial.endpoint;

    // Run the real Node app with the injected environment a launched command receives.
    const port = 40000 + Math.floor(Math.random() * 20000);
    const env: Record<string, string> = { ...(process.env as Record<string, string>), PORT: String(port),
      OTEL_EXPORTER_OTLP_ENDPOINT: endpoint,
      OTEL_EXPORTER_OTLP_TRACES_ENDPOINT: `${endpoint}/v1/traces`,
      OTEL_EXPORTER_OTLP_LOGS_ENDPOINT: `${endpoint}/v1/logs` };
    app = Bun.spawn(["node", "--import", "./instrumentation.ts", "src/server.ts"], { cwd: appDir, env, stdout: "pipe", stderr: "pipe" });
    let appOut = "";
    const drain = async (s: ReadableStream<Uint8Array>) => { for await (const c of s) appOut += new TextDecoder().decode(c); };
    void drain(app.stdout as ReadableStream<Uint8Array>); void drain(app.stderr as ReadableStream<Uint8Array>);
    const deadline = Date.now() + 20_000;
    while (!appOut.includes("listening") && Date.now() < deadline) yield* Effect.sleep(100);
    assert.ok(appOut.includes("listening"), `app did not start: ${appOut}`);
    const statuses: Record<string, number> = {};
    for (const path of ["/orders/42", "/", "/fail"]) {
      const r = yield* Effect.promise(() => fetch(`http://127.0.0.1:${port}${path}`));
      statuses[path] = r.status; yield* Effect.promise(() => r.text());
    }
    console.log("app responses:", JSON.stringify(statuses));
    yield* Effect.sleep(300);
    app.kill("SIGINT"); // instrumentation.ts: sdk.shutdown() flushes, then exit 0
    const code = yield* Effect.promise(() => app!.exited);
    console.log("app exit code", code, "app output:\n" + appOut.trim());
    check("app exported and exited cleanly after flush", code === 0, { code });

    // Assertions through traces.watch (first page of a fresh watch) and traces.get, as ⌘5 reads them.
    let page: any;
    for (let i = 0; i < 50; i++) { page = yield* firstPage(); if (page.total >= 3) break; yield* Effect.sleep(200); }
    console.log("watch page traces:", JSON.stringify(page.traces.map((t: any) => ({ id: t.id, root: t.root, service: t.service, spans: t.spans, errors: t.errors, status: t.status, sessionId: t.sessionId, turn: t.turn, durationMs: t.durationMs })), null, 1));
    check("watch lists 3 traces", page.total === 3 && page.traces.length === 3, { total: page.total });
    check("every trace service.name is orders-api", page.traces.every((t: any) => t.service === "orders-api"));
    check("every trace attributed to the session", page.traces.every((t: any) => t.sessionId === TEST_SESSION));
    const failing = page.traces.find((t: any) => t.root.includes("GET") && t.errors > 0);
    const errored = page.traces.filter((t: any) => t.status === "error");
    check("exactly one trace marked error (the /fail request)", errored.length === 1 && failing !== undefined && errored[0].id === failing.id, errored.map((t: any) => ({ id: t.id, root: t.root, errors: t.errors })));
    const errorPage: any = yield* firstPage({ status: "error" });
    check("status:error filter returns only the failing trace", errorPage.total === 1 && errorPage.traces[0]?.id === failing?.id, { total: errorPage.total });

    const detail: any = yield* rpc["traces.get"]({ sessionId: TEST_SESSION, traceId: failing.id.slice(0, 8) });
    console.log("traces.get spans:", JSON.stringify(detail.spans.map((s: any) => ({ id: s.id, parent: s.parentId, name: s.name, kind: s.kind, status: s.status, statusMessage: s.statusMessage, ms: s.durationMs, polaris: s.polaris })), null, 1));
    check("traces.get returns the error trace with >=2 spans", detail.summary.id === failing.id && detail.spans.length >= 2, { spans: detail.spans.length });
    check("spans carry checkout + workspace + session attribution", detail.spans.every((s: any) => s.polaris.checkout === workspace && s.polaris.workspaceId === TEST_WORKSPACE && s.polaris.sessionId === TEST_SESSION), detail.spans[0]?.polaris);
    const charge = detail.spans.find((s: any) => s.name === "orders.charge");
    check("orders.charge span has error status", charge?.status === "error", { status: charge?.status, events: charge?.events.map((e: any) => e.name) });
    const server = detail.spans.find((s: any) => s.kind === "server");
    const spanDetail: any = yield* rpc["traces.span"]({ sessionId: TEST_SESSION, traceId: failing.id, spanId: server.id.slice(0, 8) });
    check("traces.span returns server span attributes", spanDetail.span.id === server.id && spanDetail.span.attributes["http.response.status_code"] !== undefined, { name: spanDetail.span.name, status_code: spanDetail.span.attributes["http.response.status_code"], route: spanDetail.span.attributes["url.path"] ?? spanDetail.span.attributes["http.target"] });
    const okTrace = page.traces.find((t: any) => t.root.includes("orders") || t !== failing);
    const okDetail: any = yield* rpc["traces.get"]({ sessionId: TEST_SESSION, traceId: page.traces.find((t: any) => t.errors === 0 && t.spans > 1)?.id ?? okTrace.id });
    console.log("ok trace spans:", okDetail.spans.map((s: any) => `${s.name}(${s.kind},${s.status})`).join(", "));

    // MCP HTTP as the session's binding. Any Constellation RPC loads the lazy MCP listener.
    yield* rpc["constellation.defaults.get"]({}).pipe(Effect.ignore);
    const lsof = Bun.spawnSync(["lsof", "-nP", "-a", "-p", String(daemon!.pid), "-iTCP", "-sTCP:LISTEN", "-Fn"]).stdout.toString();
    const ports = [...lsof.matchAll(/n127\.0\.0\.1:(\d+)/g)].map((m) => Number(m[1])).filter((p) => p !== 4319);
    console.log("daemon loopback listeners:", ports);
    let origin: string | undefined;
    for (const p of ports) {
      const r = yield* Effect.promise(() => fetch(`http://127.0.0.1:${p}/mcp/${mcpToken}`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ jsonrpc: "2.0", id: 0, method: "tools/list" }) }).catch(() => null));
      if (r?.ok) { const body: any = yield* Effect.promise(() => r.json()); if (body?.result?.tools) { origin = `http://127.0.0.1:${p}`; console.log("MCP tools:", body.result.tools.map((t: any) => t.name).join(",")); break; } }
    }
    check("found MCP HTTP endpoint for the session token", origin !== undefined, { origin });
    const call = async (name: string, args: unknown) => {
      const r = await fetch(`${origin}/mcp/${mcpToken}`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ jsonrpc: "2.0", id: name, method: "tools/call", params: { name, arguments: args } }) });
      return ((await r.json()) as any).result;
    };
    const listed: any = yield* Effect.promise(() => call("traces", { turn: "all" }));
    console.log("MCP traces text:\n" + listed.content[0].text);
    const ids = (listed.structuredContent?.traces ?? []).map((t: any) => t.id).sort();
    check("MCP traces lists the same 3 traces", listed.isError !== true && JSON.stringify(ids) === JSON.stringify(page.traces.map((t: any) => t.id).sort()), { ids });
    const listedDefault: any = yield* Effect.promise(() => call("traces", {}));
    console.log("MCP traces (default current turn) text:\n" + listedDefault.content[0].text);
    const tree: any = yield* Effect.promise(() => call("trace", { id: failing.id.slice(0, 8) }));
    console.log("MCP trace text:\n" + tree.content[0].text);
    check("MCP trace returns the waterfall", tree.isError !== true && tree.content[0].text.includes("orders.charge") && tree.content[0].text.includes(failing.root), { firstLine: tree.content[0].text.split("\n")[0] });
    const full: any = yield* Effect.promise(() => call("span", { trace: failing.id, span: charge.id.slice(0, 8) }));
    console.log("MCP span text:\n" + full.content[0].text);
    const sp = full.structuredContent?.span;
    check("MCP span returns an attribute and attribution", full.isError !== true && sp?.id === charge.id && sp?.polaris?.checkout === workspace && sp?.polaris?.sessionId === TEST_SESSION && Object.keys(sp?.attributes ?? {}).length > 0, { attributes: sp?.attributes, status: sp?.status });

    if (keep) {
      console.log(`KEEP: daemon socket ${daemon!.socketPath}; touch /tmp/g3e2e/stop to finish`);
      while (!(yield* Effect.promise(() => Bun.file("/tmp/g3e2e/stop").exists()))) yield* Effect.sleep(1000);
    }
  })));
} catch (error) {
  console.error("ERROR", error);
  check("script completed without exception", false, String(error));
  if (daemon) console.error("daemon logs tail:\n" + daemon.logs().slice(-4000));
} finally {
  if (app && app.exitCode === null) app.kill("SIGKILL");
  await daemon?.stop();
  await writeFile("/tmp/g3e2e/daemon.log", daemon?.logs() ?? "");
  await rm(home, { recursive: true, force: true });
  await rm(workspace, { recursive: true, force: true });
  console.log("cleaned up", home, workspace);
  console.log("SUMMARY", JSON.stringify(results.map((r) => `${r.pass ? "PASS" : "FAIL"} ${r.name}`), null, 1));
  process.exit(results.every((r) => r.pass) ? 0 : 1);
}
